import SwiftUI import WebKit struct WebView: UIViewRepresentable { func makeUIView(context: Context) -> WKWebView { let config = WKWebViewConfiguration() config.allowsInlineMediaPlayback = true let webView = WKWebView(frame: .zero, configuration: config) if let filePath = Bundle.main.path(forResource: "index", ofType: "html") { let fileURL = URL(fileURLWithPath: filePath) webView.loadFileURL(fileURL, allowingReadAccessTo: fileURL.deletingLastPathComponent()) } return webView } func updateUIView(_ uiView: WKWebView, context: Context) {} } struct ContentView: View { var body: some View { WebView() .edgesIgnoringSafeArea(.all) } }
